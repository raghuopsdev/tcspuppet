Docker install - puppet master
===================================
sudo apt-get update
sudo apt-get install -y docker.io
sudo systemctl enable --now docker

sudo mkdir -p /opt/puppetlabs/server/data /etc/puppetlabs/puppet /etc/puppetlabs/puppetserver

sudo docker run -d \
  --name puppetserver \
  -p 8140:8140 \
  --hostname puppet.example.com \
  puppet/puppetserver:latest


sudo docker logs -f puppetserver

curl -k https://localhost:8140/status/v1/simple

Docker - Puppet Agent
======================

sudo docker network create puppet-net
docker network ls // To confirm that puppet-net is listed.

sudo docker run -d \
  --name puppetserver \
  --net puppet-net \
  --hostname puppetserver.local \
  -p 8140:8140 \
  puppet/puppetserver:latest

docker logs -f puppetserver
// And wait until you see the message: Puppet Server has successfully started and is now ready to handle requests

sudo docker run -d \
  --name puppet-agent \
  --net puppet-net \
  --hostname agent.local \
  -e PUPPET_SERVER=puppet \
  puppet/puppet-agent:latest \
  puppet agent --verbose --onetime --no-daemonize --server puppet
